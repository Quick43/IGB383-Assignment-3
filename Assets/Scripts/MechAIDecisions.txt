tree("Root")
    parallel
        tree "BehaviourRoot"
        tree "WeaponRoot"

tree("BehaviourRoot")
    repeat
        fallback
            tree "Fleeing"
            tree "Attacking"
            tree "Pursuing"
            tree "Roaming"

tree("Fleeing")
    while StatusCheck()
        repeat
            Flee()

tree("Attacking")
    while
        sequence
            HasAttackTarget()
            TargetLOS()
            not StatusCheck()
        repeat
            sequence
                CombatAim()
                fallback
                    sequence
                        ShouldRetreat()
                        CombatRetreat()
                    sequence
                        ShouldApproach()
                        CombatApproach()
                    sequence
                        ShouldReposition()
                        CombatReposition()
                    sequence
                        ShouldStrafe()
                        CombatStrafe()
                    CombatHold()

tree("Pursuing")
    while 
        sequence
            HasAttackTarget()
            not TargetLOS()
            not StatusCheck()
            HasRecentTargetMemory()
        repeat
            Pursue()

tree("Roaming")
    while not HasAttackTarget()
        repeat
            Roam()
